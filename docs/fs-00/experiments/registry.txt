const fs = require('node:fs');
(async () => {
 const names = ['web-tree-sitter','tree-sitter-python','tree-sitter-java','tree-sitter-kotlin','tree-sitter-swift','tree-sitter-objc','@tree-sitter-grammars/tree-sitter-kotlin','vite','react','react-dom','next','react-router','react-native','metro','expo','expo-router','@react-navigation/native','@react-navigation/native-stack','axios','hermes-parser'];
 const result = {retrievedUtc:new Date().toISOString(),npm:{},pypi:{}};
 for (const name of names) {
  try {
   const response = await fetch('https://registry.npmjs.org/'+encodeURIComponent(name));
   if (!response.ok) throw new Error('HTTP '+response.status);
   const data = await response.json();
   const stable = Object.keys(data.versions).filter(v=>/^\d+\.\d+\.\d+$/.test(v));
   const prefixes = {vite:['7.','8.'],react:['18.3.','19.2.'],next:['15.','16.3.'], 'react-router':['7.'], 'react-native':['0.83.','0.85.'],metro:['0.83.','0.84.'],expo:['55.','56.'], 'expo-router':['55.','56.'], '@react-navigation/native':['7.'], '@react-navigation/native-stack':['7.'], 'web-tree-sitter':['0.25.','0.26.','0.27.']};
   const selected = [...new Set([data['dist-tags'].latest,...(prefixes[name]??[]).map(p=>stable.filter(v=>v.startsWith(p)).sort((a,b)=>{const x=a.split('.').map(Number),y=b.split('.').map(Number);for(let i=0;i<3;i++)if(x[i]!==y[i])return y[i]-x[i];return 0;})[0]).filter(Boolean)])];
   result.npm[name]={source:'https://registry.npmjs.org/'+encodeURIComponent(name),tags:data['dist-tags'],selected:selected.map(v=>({version:v,time:data.time?.[v],engines:data.versions[v].engines,peers:data.versions[v].peerDependencies,dependencies:data.versions[v].dependencies,dist:data.versions[v].dist,repository:data.versions[v].repository,license:data.versions[v].license}))};
  } catch(error){result.npm[name]={error:String(error)};}
 }
 for(const name of ['Django','djangorestframework']) {
  const response = await fetch('https://pypi.org/pypi/'+name+'/json'); const data=await response.json();
  const prefixes=name==='Django'?['5.2.','6.0.']:['3.16.','3.17.'];
  const stable=Object.keys(data.releases).filter(v=>/^\d+\.\d+\.\d+$/.test(v));
  const selected=[...new Set([data.info.version,...prefixes.map(p=>stable.filter(v=>v.startsWith(p)).sort((a,b)=>Number(b.split('.')[2])-Number(a.split('.')[2]))[0]).filter(Boolean)])];
  result.pypi[name]={source:'https://pypi.org/pypi/'+name+'/json',selected:[]};
  for(const version of selected){const r=await fetch('https://pypi.org/pypi/'+name+'/'+version+'/json');const d=await r.json();result.pypi[name].selected.push({version,requiresPython:d.info.requires_python,requiresDist:d.info.requires_dist,files:d.urls.map(f=>({filename:f.filename,sha256:f.digests.sha256,uploadTime:f.upload_time_iso_8601,yanked:f.yanked}))});}
 }
 fs.writeFileSync('docs/fs-00/evidence/registry.json',JSON.stringify(result,null,2)+'\n');
 console.log(JSON.stringify({npm:Object.fromEntries(Object.entries(result.npm).map(([k,v])=>[k,v.selected?.map(x=>x.version)??v.error])),pypi:result.pypi},null,2));
})().catch(e=>{console.error(e);process.exitCode=1});
