const fs=require('fs'),path=require('path'),crypto=require('crypto');
const base=path.resolve('docs/fs-00/experiments/generated');
const artifacts=[];
function walk(dir){for(const e of fs.readdirSync(dir,{withFileTypes:true})){const p=path.join(dir,e.name);if(e.isSymbolicLink())continue;if(e.isDirectory())walk(p);else if(e.name.endsWith('.wasm')||e.name==='package-lock.json'||e.name==='LICENSE'||e.name==='LICENSE.txt'){const b=fs.readFileSync(p);artifacts.push({path:path.relative(base,p).replaceAll('\\','/'),bytes:b.length,sha256:crypto.createHash('sha256').update(b).digest('hex')});}}}
walk(base);
for(const p of ['node_modules/.pnpm/hermes-parser@0.25.1/node_modules/hermes-parser','node_modules/.pnpm/hermes-parser@0.25.1/node_modules/hermes-estree']){if(fs.existsSync(p)){const pkg=JSON.parse(fs.readFileSync(path.join(p,'package.json'),'utf8'));artifacts.push({path:p+'/package.json',package:pkg.name,version:pkg.version,license:pkg.license});for(const e of fs.readdirSync(path.join(p,'dist'),{withFileTypes:true})){if(e.isFile()&&/HermesParser|wasm/i.test(e.name)){const filename=path.join(p,'dist',e.name),b=fs.readFileSync(filename);artifacts.push({path:filename.replaceAll('\\','/'),bytes:b.length,sha256:crypto.createHash('sha256').update(b).digest('hex')});}}}}
const locks=[];for(const name of fs.readdirSync(base)){const p=path.join(base,name,'package-lock.json');if(fs.existsSync(p)){const lock=JSON.parse(fs.readFileSync(p,'utf8'));locks.push({experiment:name,lockfileVersion:lock.lockfileVersion,packages:Object.entries(lock.packages).map(([k,v])=>({path:k,version:v.version,resolved:v.resolved,integrity:v.integrity,license:v.license,hasInstallScript:v.hasInstallScript}))});}}
fs.writeFileSync('docs/fs-00/evidence/parser-assets.json',JSON.stringify({hostNode:process.version,runtime:'web-tree-sitter 0.25.10',artifacts,locks,note:'Experiment installed packages removed after inventory. No native install scripts executed; production dependency approval remains separate.'},null,2)+'\n');
console.log('Recorded '+artifacts.length+' assets/license/lock records');
